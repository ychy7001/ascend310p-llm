cmake_minimum_required(VERSION 3.20)
project(qwen36_mindieq_grouped LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
include(${PROJECT_REPO}/experimental/ascend_c8_kernel/cmake/config_envs.cmake)
include(${PROJECT_REPO}/experimental/ascend_c8_kernel/cmake/config_ascend.cmake)
ascendc_library(grouped_device STATIC grouped.cpp activation_quant.cpp)
add_library(qwen36_mindieq_grouped SHARED register.cpp)
target_include_directories(qwen36_mindieq_grouped PRIVATE ${TORCH_DIR}/include ${TORCH_DIR}/include/torch/csrc/api/include ${TORCH_NPU_DIR}/include ${ASCEND_HOME_PATH}/include)
target_link_directories(qwen36_mindieq_grouped PRIVATE ${TORCH_DIR}/lib ${TORCH_NPU_DIR}/lib ${ASCEND_HOME_PATH}/lib64)
target_link_libraries(qwen36_mindieq_grouped PRIVATE grouped_device torch_npu torch torch_cpu c10 ascendcl)
set_target_properties(qwen36_mindieq_grouped PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/output)
