cmake_minimum_required(VERSION 3.20)
project(c8_mixed_mmad_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(ASCEND_CANN_PACKAGE_PATH $ENV{ASCEND_HOME_PATH})
include(${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
option(PROBE_FP16_CONTROL "Compile the supported half x half control" OFF)
if(PROBE_FP16_CONTROL)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/mixed_mmad.cpp" PROBE_SOURCE)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/control_mmad.cpp" "#define PROBE_FP16_CONTROL 1\n${PROBE_SOURCE}")
  ascendc_library(c8_mixed_mmad_probe STATIC "${CMAKE_CURRENT_BINARY_DIR}/control_mmad.cpp")
else()
  ascendc_library(c8_mixed_mmad_probe STATIC mixed_mmad.cpp)
endif()
